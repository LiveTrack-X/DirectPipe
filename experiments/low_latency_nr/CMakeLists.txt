cmake_minimum_required(VERSION 3.22)
project(DirectPipeStftExperiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(stft-nr-tests tests.cpp)
add_executable(stft-nr-probe probe.cpp)
if(MSVC)
    target_compile_options(stft-nr-tests PRIVATE /W4 /utf-8 /fp:precise)
    target_compile_options(stft-nr-probe PRIVATE /W4 /utf-8 /fp:precise)
else()
    target_compile_options(stft-nr-tests PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(stft-nr-probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME stft-nr-tests COMMAND stft-nr-tests)
